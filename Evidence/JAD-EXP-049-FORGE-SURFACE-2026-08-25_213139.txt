  144:     const auto beats =
  145:         transport.getPositionInBeats();

  154:         static_cast<int> (
  155:             std::floor (beats)

  158:     const auto bar =
  159:         wholeBeat / numerator + 1;
  160: 
  161:     const auto beat =
  162:         wholeBeat % numerator + 1;
  163: 
  164:     const auto fraction =
  165:         static_cast<int> (
  166:             std::floor (
  167:                 (
  168:                     beats
  169:                     - std::floor (beats)
  170:                 )
  171:                 * 1000.0
  172:             )
  173:         );
  174: 
  175:     positionLabel.setText (
  176:         juce::String (bar)
  177:             + " | "
  178:             + juce::String (beat)
  179:             + " | "
